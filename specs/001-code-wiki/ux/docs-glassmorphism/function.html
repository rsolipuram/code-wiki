authenticateUser

Authenticates a user with their credentials, validates against the database, and generates a JWT token for session management. Supports both traditional username/password authentication and OAuth providers.

📁 src/auth/authenticator.ts
📏 Lines 42-87
👤 Public API

Signature

async function authenticateUser( username: string, password: string, options?: AuthOptions ): Promise<AuthResult>

Parameters

Name Type Description Required
username string The username or email address of the user attempting to authenticate Yes
password string The user's password in plain text (will be hashed internally for comparison) Yes
options AuthOptions Optional authentication settings including session duration, device fingerprint, and MFA requirements No

Return Value

Type Description
Promise<AuthResult> Returns a promise that resolves to an AuthResult object containing:
  • success (boolean) - Whether authentication succeeded
  • token (string | null) - JWT token if successful
  • user (User | null) - User object with profile information
  • error (string | null) - Error message if authentication failed

Usage Examples

Basic Authentication
import { authenticateUser } from './auth/authenticator';

async function handleLogin(username: string, password: string) {
  const result = await authenticateUser(username, password);

  if (result.success) {
    console.log('Login successful!', result.user);
    return result.token;
  } else {
    console.error('Login failed:', result.error);
    return null;
  }
}
With Custom Options
const result = await authenticateUser(
  'user@example.com',
  'securePassword123',
  {
    sessionDuration: 3600, // 1 hour in seconds
    requireMFA: true,
    deviceFingerprint: getDeviceFingerprint()
  }
);

if (result.success && result.requiresMFA) {
  // Redirect to MFA verification page
  redirectToMFA(result.mfaToken);
}
Error Handling
try {
  const result = await authenticateUser(username, password);

  if (!result.success) {
    switch (result.error) {
      case 'INVALID_CREDENTIALS':
        showError('Username or password is incorrect');
        break;
      case 'ACCOUNT_LOCKED':
        showError('Account has been locked due to too many failed attempts');
        break;
      case 'ACCOUNT_DISABLED':
        showError('This account has been disabled');
        break;
    }
  }
} catch (error) {
  console.error('Authentication error:', error);
}

Source Code

async function authenticateUser(
  username: string,
  password: string,
  options: AuthOptions = {}
): Promise<AuthResult> {
  // Validate input parameters
  if (!username || !password) {
    return {
      success: false,
      error: 'INVALID_INPUT',
      token: null,
      user: null
    };
  }

  // Find user in database
  const user = await findUserByUsername(username);
  if (!user) {
    await logAuthAttempt(username, 'FAILED', 'USER_NOT_FOUND');
    return {
      success: false,
      error: 'INVALID_CREDENTIALS',
      token: null,
      user: null
    };
  }

  // Check if account is locked or disabled
  if (user.status === 'LOCKED') {
    return { success: false, error: 'ACCOUNT_LOCKED', token: null, user: null };
  }

  // Validate password
  const isValid = await validateCredentials(user, password);
  if (!isValid) {
    await logAuthAttempt(username, 'FAILED', 'INVALID_PASSWORD');
    return {
      success: false,
      error: 'INVALID_CREDENTIALS',
      token: null,
      user: null
    };
  }

  // Generate JWT token
  const token = await generateToken(user, options);
  await logAuthAttempt(username, 'SUCCESS');

  return {
    success: true,
    token,
    user,
    error: null
  };
}
